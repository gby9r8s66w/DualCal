Embeddable compact widget

One custom element, zero network at runtime. Attributes map to settings; events: date-change, settings-change, holiday-select, error. Works inside any host page after the module is cached.

<script type="module" src="./js/widget.js"></script>
<dual-calendar-sky-widget
  app-language="en" gregorian-language="en" hijri-language="ar"
  primary-calendar="gregorian" hijri-method="islamic-umalqura"
  rollover="sunset" sunset-offset-minutes="0"
  country="SA" theme="auto">
</dual-calendar-sky-widget>
